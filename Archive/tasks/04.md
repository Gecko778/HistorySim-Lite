### 五、知识点

BFS 适用于边成本相同的地图，可用于“最多移动 N 步”。Dijkstra 适用于非负权重图，可用于地形成本不同的行军路线。教学地图只有 30 个地区，但仍要求学生说明 O(V+E) 与 O((V+E)logV) 的复杂度意义。

### 六、示例框架代码（TS 版）

```ts
export function getReachableRegions(regions: Record<string, Region>, start: string, maxSteps: number): string[] {
  const visited = new Set<string>([start]);
  const queue: Array<{ id: string; d: number }> = [{ id: start, d: 0 }];
  while (queue.length) {
    const cur = queue.shift()!;
    if (cur.d >= maxSteps) continue;
    for (const n of regions[cur.id].neighbours) {
      if (!visited.has(n)) { visited.add(n); queue.push({ id: n, d: cur.d + 1 }); }
    }
  }
  return [...visited].filter(id => id !== start);
}

export function findLowestCostPath(regions: Record<string, Region>, start: string, target: string, cost: Record<string, number>) {
  const dist: Record<string, number> = {}; const prev: Record<string, string | undefined> = {};
  const unvisited = new Set(Object.keys(regions));
  for (const id of unvisited) dist[id] = Number.POSITIVE_INFINITY;
  dist[start] = 0;
  while (unvisited.size) {
    const current = [...unvisited].sort((a,b) => dist[a] - dist[b])[0];
    if (current === target || dist[current] === Number.POSITIVE_INFINITY) break;
    unvisited.delete(current);
    for (const n of regions[current].neighbours) {
      const alt = dist[current] + (cost[regions[n].terrain] ?? 1);
      if (alt < dist[n]) { dist[n] = alt; prev[n] = current; }
    }
  }
  if (dist[target] === Number.POSITIVE_INFINITY) return null;
  const path: string[] = []; let cur: string | undefined = target;
  while (cur) { path.unshift(cur); cur = prev[cur]; }
  return { path, cost: dist[target] };
}
```

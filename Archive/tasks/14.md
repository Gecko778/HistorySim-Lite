### 五、知识点

教学版可将完整 GameState 保存为 JSON，降低数据库拆表难度。Repository 模式可避免 UI 和 Core 直接依赖具体数据库 API，便于替换持久化方案。

### 六、示例框架代码（TS 版）

```ts
export interface SaveSummary { slotName: string; turn: number; date: string; updatedAt: string; }
export interface SaveRepository {
  save(slotName: string, state: GameState): Promise<void>;
  load(slotName: string): Promise<GameState>;
  list(): Promise<SaveSummary[]>;
}

export class InMemorySaveRepository implements SaveRepository {
  private saves = new Map<string, GameState>();
  async save(slotName: string, state: GameState): Promise<void> { this.saves.set(slotName, structuredClone(state)); }
  async load(slotName: string): Promise<GameState> {
    const state = this.saves.get(slotName);
    if (!state) throw new Error(`Save not found: ${slotName}`);
    return structuredClone(state);
  }
  async list(): Promise<SaveSummary[]> {
    return [...this.saves.entries()].map(([slotName, s]) => ({ slotName, turn: s.turn, date: s.date, updatedAt: new Date().toISOString() }));
  }
}

// SQLite 表：saves(slot_name, turn, game_date, random_seed, state_json, created_at, updated_at)
```

### 六、示例框架代码（TS 版）

```ts
export type PersonStat = "command" | "strength" | "intelligence" | "politics" | "charisma";

export function rankPersonsByStat(persons: Person[], stat: PersonStat): Person[] {
  return [...persons].filter(p => p.status === "active").sort((a, b) => b[stat] - a[stat]);
}

export function recruitTroops(region: Region, faction: Faction, amount: number) {
  const orderModifier = Math.max(0.2, region.publicOrder / 100);
  const maxRecruit = Math.floor(region.population * 0.05 * orderModifier);
  const goldCost = Math.ceil(amount * 0.2);
  const foodCost = Math.ceil(amount * 0.3);
  if (amount <= 0 || amount > maxRecruit) return { accepted: false, region, faction, message: "Invalid recruit amount." };
  if (faction.gold < goldCost || faction.food < foodCost) return { accepted: false, region, faction, message: "Not enough resources." };
  return {
    accepted: true,
    message: `Recruited ${amount} troops in ${region.name}.`,
    region: { ...region, population: region.population - amount, troops: region.troops + amount, publicOrder: Math.max(0, region.publicOrder - Math.ceil(amount / 1000)) },
    faction: { ...faction, gold: faction.gold - goldCost, food: faction.food - foodCost },
  };
}
```

### 五、知识点

推荐依赖方向为 ui -> app -> core，rendering 读取 core types，persistence 通过 repository 接口保存 core state。Core 不依赖 React、PixiJS、Tauri。架构的价值在于降低变化成本。

### 六、示例框架代码（TS 版）

```ts
export interface GameService {
  getState(): GameState;
  submitCommand(command: Command): void;
  advance(): GameState;
}

export class LocalGameService implements GameService {
  private state: GameState;
  private pending: Command[] = [];
  constructor(initialState: GameState) { this.state = initialState; }
  getState(): GameState { return this.state; }
  submitCommand(command: Command): void { this.pending.push(command); }
  advance(): GameState {
    this.state = advanceTurn(this.state, this.pending);
    this.pending = [];
    return this.state;
  }
}

export interface SaveRepository {
  save(slotName: string, state: GameState): Promise<void>;
  load(slotName: string): Promise<GameState>;
  listSlots(): Promise<string[]>;
}
```

### 一、本次主题

本次课实现 ATTACK 命令和抽象战斗结算模型，使用固定随机种子保证战斗可复现。

### 五、知识点

教学版不制作战术地图，而采用抽象战斗结算。战斗结果应能解释主要影响因素，并能用同一随机种子复现。禁止只用 Math.random() 产生不可复现结果。

### 六、示例框架代码（TS 版）

```ts
export class SeededRandom {
  constructor(private seed: number) {}
  next(): number { this.seed = (1664525 * this.seed + 1013904223) >>> 0; return this.seed / 0xffffffff; }
  between(min: number, max: number): number { return min + (max - min) * this.next(); }
  currentSeed(): number { return this.seed; }
}

export interface BattleSide { army: Army; leader: Person; region: Region; }
export function calculateBattlePower(side: BattleSide, role: "attacker" | "defender"): number {
  const troopPower = side.army.troops;
  const commandModifier = 1 + side.leader.command / 200;
  const moraleModifier = Math.max(0.5, side.army.morale / 100);
  const defenceModifier = role === "defender" ? 1 + side.region.defence / 200 : 1;
  return Math.round(troopPower * commandModifier * moraleModifier * defenceModifier);
}

export function resolveBattle(attacker: BattleSide, defender: BattleSide, seed: number) {
  const rng = new SeededRandom(seed);
  const attackerPower = Math.round(calculateBattlePower(attacker, "attacker") * rng.between(0.9, 1.1));
  const defenderPower = Math.round(calculateBattlePower(defender, "defender") * rng.between(0.9, 1.1));
  return { attackerWon: attackerPower > defenderPower, attackerPower, defenderPower, nextSeed: rng.currentSeed() };
}
```

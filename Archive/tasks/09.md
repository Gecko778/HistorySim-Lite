## 第 9 次课：军队移动、路径成本与状态机

### 一、本次主题

本次课实现 MOVE_ARMY 命令，结合路径搜索判断移动合法性，并使用状态机管理 Army 状态。

### 二、课程目标

1. 使用路径算法支持军队移动。
2. 判断路径、成本、所属和状态合法性。
3. 设计 Army 状态机。
4. 在地图上显示军队位置。

### 五、知识点

军队不是地图上的装饰，而是具有状态的实体。状态机可防止处于 fighting 或 retreating 的军队同时执行移动。移动应检查军队存在、所属势力、目标地区、路径和资源。

### 六、示例框架代码（TS 版）

```ts
export type ArmyAction = "startMove" | "arrive" | "startBattle" | "retreat" | "recover";

export function transitionArmyStatus(army: Army, action: ArmyAction): Army {
  if (army.status === "stationed" && action === "startMove") return { ...army, status: "moving" };
  if (army.status === "moving" && action === "arrive") return { ...army, status: "stationed" };
  if (army.status === "stationed" && action === "startBattle") return { ...army, status: "fighting" };
  if (army.status === "fighting" && action === "recover") return { ...army, status: "stationed" };
  if (action === "retreat") return { ...army, status: "retreating" };
  if (army.status === "retreating" && action === "recover") return { ...army, status: "stationed" };
  throw new Error(`Invalid transition: ${army.status} -> ${action}`);
}

export function moveArmy(regions: Record<string, Region>, army: Army, faction: Faction, target: string, terrainCost: Record<string, number>) {
  if (army.status !== "stationed") return { accepted: false, army, faction, message: "Army is not stationed." };
  const path = findLowestCostPath(regions, army.locationRegionId, target, terrainCost);
  if (!path) return { accepted: false, army, faction, message: "Target unreachable." };
  const foodCost = path.cost * Math.ceil(army.troops / 1000);
  if (faction.food < foodCost) return { accepted: false, army, faction, message: "Not enough food." };
  return { accepted: true, army: { ...army, locationRegionId: target }, faction: { ...faction, food: faction.food - foodCost }, path: path.path, message: "Army moved." };
}
```

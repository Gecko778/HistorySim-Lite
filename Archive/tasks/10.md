## 第 10 次课：事件系统、规则配置与效果执行

### 一、本次主题

本次课建立 Event 系统，通过配置化触发条件和效果执行机制处理随机事件、历史事件和系统事件。

### 五、知识点

事件系统将条件判断和状态改变集中处理，避免事件逻辑散落在战斗、经济和 AI 模块中。教学版可用有限枚举表达触发类型和效果类型，不要求完整规则语言解释器。

### 六、示例框架代码（TS 版）

```ts
export interface EventDefinition {
  id: string;
  name: string;
  category: "historical" | "random" | "system";
  triggerType: "LOW_FOOD_LOW_ORDER" | "TURN_REACHED" | "LOW_LOYALTY";
  effectType: "REGION_VALUE_CHANGE" | "PERSON_STATUS_CHANGE" | "FACTION_RESOURCE_CHANGE";
  params: Record<string, unknown>;
  logTemplate: string;
  once?: boolean;
}

export function isEventTriggered(event: EventDefinition, state: GameState): boolean {
  if (event.triggerType === "LOW_FOOD_LOW_ORDER") {
    return Object.values(state.regions).some(r => r.food < r.population * 0.05 && r.publicOrder < 50);
  }
  if (event.triggerType === "TURN_REACHED") return state.turn >= Number(event.params.turn ?? 0);
  return false;
}

export function resolveEvents(state: GameState, events: EventDefinition[]): GameState {
  const next = structuredClone(state);
  for (const e of events) {
    if (!isEventTriggered(e, next)) continue;
    next.logs.push({ turn: next.turn, category: "event", message: `${e.name} triggered.` });
  }
  return next;
}
```

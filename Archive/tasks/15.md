### 五、知识点

测试不是只验证界面能点击。核心是证明模拟内核可运行、可复现、可审计。模拟审计应检查负数、无效引用、非法归属、军队位置错误等问题。

### 六、示例框架代码（TS 版）

```ts
export interface AuditIssue { level: "error" | "warning"; message: string; }

export function auditGameState(state: GameState): AuditIssue[] {
  const issues: AuditIssue[] = [];
  for (const r of Object.values(state.regions)) {
    if (!state.factions[r.ownerFactionId]) issues.push({ level: "error", message: `${r.id} invalid owner.` });
    if (r.population < 0 || r.food < 0 || r.troops < 0) issues.push({ level: "error", message: `${r.id} has negative values.` });
  }
  for (const a of Object.values(state.armies)) {
    if (!state.regions[a.locationRegionId]) issues.push({ level: "error", message: `${a.id} invalid location.` });
    if (a.troops < 0) issues.push({ level: "error", message: `${a.id} negative troops.` });
  }
  return issues;
}
```
